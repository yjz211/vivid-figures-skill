  26     r_prev = ring_levels[k-1] if k > 0 else 0
  27     if k % 2 == 0:

  33 ax.set_xticks(angles[:-1])

  35 ax.tick_params(axis='x', pad=18)
  36 
  37 for i, (name, vals) in enumerate(methods.items()):
