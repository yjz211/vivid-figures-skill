1 import numpy as np
   2 import matplotlib.pyplot as plt
   3 from _utils.plot_utils import setup_style, save_fig, PALETTE, COLORS
   4 setup_style()
   5 
   6 np.random.seed(42)
   7 iters = np.arange(0, 501, 5)
   8 methods = {
   9     '遗传算法 (GA)': 100*np.exp(-0.008*iters) + 5 + np.random.normal(0,0.5,len(iters))*0.5,
  10     '粒子群 (PSO)': 100*np.exp(-0.012*iters) + 3 + np.random.normal(0,0.4,len(iters))*0.5,
  11     '模拟退火 (SA)': 100*np.exp(-0.006*iters) + 8 + np.random.normal(0,0.6,len(iters))*0.5,
  12     '本文算法': 100*np.exp(-0.018*iters) + 1.5 + np.random.normal(0,0.2,len(iters))*0.3,
  13 }
  14 
  15 fig, ax = plt.subplots(figsize=(7, 4.5))
  16 highlight_index = min(3, len(methods) - 1)  # 重点系列的位置，与显示名称无关
  17 ax.grid(True, linestyle='--', alpha=0.15, color=COLORS['grid']); ax.set_axisbelow(True)
  18 
  19 _end_xs, _end_ys, _end_texts, _end_colors = [], [], [], []
  20 for i, (name, vals) in enumerate(methods.items()):
  21     lw = 2.5 if i == highlight_index else 1.5
  22     alpha = 1.0 if i == highlight_index else 0.55
  23     ax.plot(iters, vals, color=PALETTE[i], linewidth=lw, label=name, alpha=alpha, zorder=3)
  24     _end_xs.append(iters[-1]); _end_ys.append(vals[-1])
  25     _end_texts.append(f'{vals[-1]:.1f}'); _end_colors.append(PALETTE[i])
  26 # ★ 终点标注用 smart_labels 防重叠
  27 from _utils.plot_utils import save_fig, smart_labels
  28 smart_labels(ax, _end_xs, _end_ys, _end_texts, colors=_end_colors, fontsize=8, offset=(8, 0))
  29 
  30 # 本文算法的下方填充区域（强调单独）
  31 ours = list(methods.values())[highlight_index]
  32 ax.fill_between(iters, ours, alpha=0.10, color=PALETTE[highlight_index], linewidth=0, zorder=1)
  33 
  34 # 收敛点标注
  35 diff = np.abs(np.diff(ours))
  36 conv_idx = np.argmax(diff < 0.15) + 1
  37 ax.annotate('收敛点', xy=(iters[conv_idx], ours[conv_idx]),
  38             xytext=(iters[conv_idx]+60, ours[conv_idx]+12),
  39             fontsize=10, color=COLORS['down'], fontweight='bold',
  40             arrowprops=dict(arrowstyle='->', color=COLORS['down'], lw=1.5), zorder=5)
  41 ax.scatter([iters[conv_idx]], [ours[conv_idx]], color=COLORS['down'], s=50, zorder=5, edgecolors='white')
  42 
  43 ax.set_xlabel('迭代次数', fontsize=11); ax.set_ylabel('目标函数值', fontsize=11)
  44 # ★ 图例用 loc='best' 自适应，不要硬编码 'upper right'（避免数据在右上角时遮挡图例）
  45 ax.legend(frameon=False, labelspacing=0.35, handlelength=1.6, fontsize=9, loc='best')
  46 ax.set_xlim(0, 520)
  47 ax.spines['top'].set_visible(False)
  48 ax.spines['right'].set_visible(False)
  49 fig.tight_layout()
  50 save_fig(fig, 'figures/fig_convergence.pdf')
